---
title: Next.js app router
description: A client host in the root layout that closes dialogs on navigation.
---

```tsx
// app/dialogs.tsx
'use client'
import { usePathname } from 'next/navigation'
import { DialogHost } from 'react-confirm-alert'

export function Dialogs() {
  return <DialogHost dismissKey={usePathname()} />
}
```

```tsx
// app/layout.tsx (a server component)
import 'react-confirm-alert/src/react-confirm-alert.css'
import { Dialogs } from './dialogs'

export default function RootLayout({ children }) {
  return <html><body>{children}<Dialogs /></body></html>
}
```

The package starts with `'use client'`, so server components can import it. Call `confirm()` from client code: on the server it rejects.
